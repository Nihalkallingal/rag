chunking.py Core Loads corpus · splits documents into Chunk objects 
"""
Step 2 — Chunking.

Turns raw documents into retrieval-ready *chunks*, each carrying provenance
metadata (source file + section header) so later steps can produce and VERIFY
citations.

Run:
    .venv/Scripts/python.exe src/chunking.py
"""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_text_splitters import MarkdownHeaderTextSplitter


CORPUS_DIR = Path(__file__).resolve().parent.parent / "data" / "corpus"

CHUNK_SIZE    = 800
CHUNK_OVERLAP = 120


@dataclass
class Chunk:
    """One retrieval unit. `text` will feed BOTH the dense and sparse indexes.
    The metadata is what makes citation + verification possible later."""
    text:        str
    source:      str          # which file this came from
    chunk_index: int          # position within that file
    start_char:  int          # offset in the original document
    metadata:    dict = field(default_factory=dict)


def load_corpus(corpus_dir: Path = CORPUS_DIR) -> list[tuple[str, str]]:
    """Ingestion layer: read every .md/.txt file in `corpus_dir` from disk.

    Returns a list of (filename, full_text) pairs. The filename travels with the
    text so provenance is never lost — it later becomes the citation label.
    """
    corpus = []
    for path in corpus_dir.glob("*"):
        if path.suffix.lower() in {".md", ".txt"}:
            corpus.append((path.name, path.read_text(encoding="utf-8")))
    return corpus


def chunk_document(filename: str, text: str) -> list[Chunk]:
    """Split ONE document into retrieval-ready chunks using two-stage splitting.

    Stage 1 (structure): MarkdownHeaderTextSplitter cuts on #/##/### headers.
    Stage 2 (size): RecursiveCharacterTextSplitter further splits large sections.
    """
    md_splitter = MarkdownHeaderTextSplitter(
        headers_to_split_on=[("#", "h1"), ("##", "h2"), ("###", "h3")])
    md_docs = md_splitter.split_text(text)
    char_splitter = RecursiveCharacterTextSplitter(
        chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP, add_start_index=True)
    chunks = []
    for md_doc in md_docs:
        sub_docs = char_splitter.create_documents([md_doc.page_content])
        for i, sub_doc in enumerate(sub_docs):
            chunks.append(Chunk(
                text=sub_doc.page_content,
                source=filename,
                chunk_index=len(chunks),
                start_char=sub_doc.metadata.get("start_index", -1),
                metadata=md_doc.metadata,
            ))
    return chunks


def build_chunks(corpus_dir: Path = CORPUS_DIR) -> list[Chunk]:
    """Top-level pipeline entry point: load every document and chunk them all."""
    all_chunks = []
    for filename, text in load_corpus(corpus_dir):
        all_chunks.extend(chunk_document(filename, text))
    return all_chunks


if __name__ == "__main__":
    chunks = build_chunks()
    print(f"Total chunks: {len(chunks)}")
    for c in chunks:
        print(f"[{c.source} #{c.chunk_index} @{c.start_char}]\n{c.text}\n")

llm_utils.py Core Exponential-backoff retry wrapper for Gemini calls 
import time
from google.api_core.exceptions import ResourceExhausted

def with_retry(fn, max_retries: int = 5, base_delay: float = 2.0):
    for attempt in range(max_retries):
        try:
            return fn()
        except ResourceExhausted:
            if attempt == max_retries - 1:
                raise
            delay = base_delay * (2 ** attempt)
            print(f"  Rate limited, waiting {delay:.0f}s...")
            time.sleep(delay)
dense_index.py Index Gemini embeddings → numpy matrix · cosine-similarity search · disk cache 
import numpy as np
from google.genai import types
from gemini_client import client
from chunking import build_chunks, Chunk
from llm_utils import with_retry
import hashlib
import pickle
from pathlib import Path


EMBEDDING_MODEL = "models/gemini-embedding-2"

CACHE_DIR  = Path(__file__).resolve().parent.parent / "data" / "index"
CACHE_PATH = CACHE_DIR / "embed_cache.pkl"

def _load_cache() -> dict:
    if CACHE_PATH.exists():
        with open(CACHE_PATH, "rb") as f:
            return pickle.load(f)
    return {}

_embed_cache = _load_cache()

def _cache_key(text: str, task_type: str) -> str:
    raw = f"{EMBEDDING_MODEL}:{task_type}:{text}"
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()

def _save_cache() -> None:
    CACHE_DIR.mkdir(parents=True, exist_ok=True)
    with open(CACHE_PATH, "wb") as f:
        pickle.dump(_embed_cache, f)


def embed_text(text: str, task_type: str) -> list[float]:
    """Call the Gemini embedding API for one piece of text, with a persistent cache.
    task_type must be RETRIEVAL_DOCUMENT at index time and RETRIEVAL_QUERY at search time."""
    key = _cache_key(text, task_type)
    if key in _embed_cache:
        return _embed_cache[key]
    result = client.models.embed_content(
        model=EMBEDDING_MODEL, contents=text,
        config=types.EmbedContentConfig(task_type=task_type),
    )
    vec = result.embeddings[0].values
    _embed_cache[key] = vec
    return vec


def build_index(chunks: list[Chunk]) -> tuple[np.ndarray, list[Chunk]]:
    """Embed all chunks as documents and return a float32 matrix of shape
    (num_chunks, embedding_dim) paired with the chunks list."""
    embeddings = []
    for chunk in chunks:
        embeddings.append(with_retry(
            lambda: embed_text(chunk.text, task_type="RETRIEVAL_DOCUMENT")))
    matrix = np.array(embeddings, dtype=np.float32)
    _save_cache()
    return matrix, chunks


def search(query: str, matrix: np.ndarray, chunks: list[Chunk],
           top_k: int = 5) -> list[tuple[float, Chunk]]:
    """Embed the query and return top_k chunks ranked by cosine similarity via dot product."""
    query_vec   = np.array(with_retry(
        lambda: embed_text(query, task_type="RETRIEVAL_QUERY")), dtype=np.float32)
    scores      = matrix @ query_vec
    top_indices = np.argsort(scores)[::-1][:top_k]
    return [(float(scores[i]), chunks[i]) for i in top_indices]

sparse_index.py Index BM25 keyword index via rank-bm25 
from rank_bm25 import BM25Okapi
from chunking import build_chunks, Chunk
import re

def tokenize(text: str) -> list[str]:
    """Lowercase and split text into word tokens.
    Must be called identically at index time and query time."""
    return re.findall(r"\w+", text.lower())

def build_bm25_index(chunks: list[Chunk]) -> BM25Okapi:
    """Tokenize all chunks and build a BM25Okapi index,
    precomputing IDF scores and document lengths."""
    tokenized = [tokenize(chunk.text) for chunk in chunks]
    return BM25Okapi(tokenized)

def search(query: str, bm25: BM25Okapi, chunks: list[Chunk],
           top_k: int = 5) -> list[tuple[float, Chunk]]:
    """Score all chunks against the query using BM25 exact-term matching.
    Chunks with no token overlap score exactly zero."""
    tokens = tokenize(query)
    scores = bm25.get_scores(tokens)
    ranked = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)
    return [(float(score), chunks[i]) for i, score in ranked[:top_k]]

fusion.py Search Reciprocal Rank Fusion — merges dense + BM25 result lists 
from chunking import Chunk
RRF_K = 60

def reciprocal_rank_fusion(
        result_lists: list[list[tuple[float, Chunk]]],
        top_k: int = 5) -> list[tuple[float, Chunk]]:
    """Fuse multiple ranked result lists using Reciprocal Rank Fusion.
    Ignores raw scores — only ranks matter, making it safe to combine
    dense and BM25 results."""
    scores:    dict[str, float] = {}
    chunk_map: dict[str, Chunk] = {}
    for results in result_lists:
        for rank, (score, chunk) in enumerate(results):
            key = f"{chunk.source}#{chunk.chunk_index}"
            scores[key]    = scores.get(key, 0.0) + 1.0 / (RRF_K + rank)
            chunk_map[key] = chunk
    ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return [(score, chunk_map[key]) for key, score in ranked[:top_k]]

from google.genai import types
from gemini_client import client
from chunking import Chunk
from llm_utils import with_retry
from concurrent.futures import ThreadPoolExecutor


RERANK_MODEL = "models/gemini-3.1-flash-lite"


def score_chunk(query: str, chunk: Chunk) -> float:
    prompt = (
        f"Rate how well the passage answers the query, on a scale of 0 to 10.\n"
        f"Respond with ONLY a single number, nothing else.\n\n"
        f"Query: {query}\n\n"
        f"Passage: {chunk.text}"
    )
    response = with_retry(lambda: client.models.generate_content(
        model=RERANK_MODEL, contents=prompt,
        config=types.GenerateContentConfig(temperature=0, max_output_tokens=5)))
    try:
        return float(response.text.strip())
    except (ValueError, AttributeError):
        return 0.0

def rerank(query: str, candidates: list[tuple[float, Chunk]],
           top_k: int = 5) -> list[tuple[float, Chunk]]:
    """Re-score each candidate by having Gemini read the query and chunk together
    (cross-encoder style), then return the top_k by fresh relevance score.
    Discards incoming RRF scores."""
    candidates_chunks = [chunk for _, chunk in candidates]
    with ThreadPoolExecutor(max_workers=5) as executor:
        scores = list(executor.map(
            lambda chunk: score_chunk(query, chunk), candidates_chunks))
    scored = list(zip(scores, candidates_chunks))
    scored.sort(key=lambda x: x[0], reverse=True)
    return scored[:top_k]

GENERATION_MODEL = "models/gemini-2.0-flash"

def format_context(chunks: list[Chunk]) -> str:
    blocks = []
    for i, chunk in enumerate(chunks, start=1):
        blocks.append(f"[{i}] ({chunk.source} #{chunk.chunk_index}) {chunk.text}")
    return "\n\n".join(blocks)

def generate_answer(query: str, chunks: list[Chunk]) -> str:
    """Generate a grounded, citation-tagged answer using ONLY the provided chunks.
    Refuses if the context lacks the answer."""
    context = format_context(chunks)
    prompt = (
        "You are a precise assistant. Answer the question using ONLY the numbered "
        "context passages below. Follow these rules strictly:\n"
        "1. Use ONLY information stated in the passages. Do NOT use prior knowledge.\n"
        "2. After each sentence, cite the passage number(s) it came from, like [1] or [1][2].\n"
        "3. If the passages do not contain the answer, reply exactly: "
        "\"I don't have enough information to answer that.\"\n\n"
        f"Context:\n{context}\n\n"
        f"Question: {query}\n\n"
        "Answer:"
    )
    response = with_retry(lambda: client.models.generate_content(
        model=GENERATION_MODEL, contents=prompt,
        config=types.GenerateContentConfig(temperature=0),
    ))
    return response.text.strip()

verification.py Generation Claim extraction + batched LLM fact-checking 
import re
from dataclasses import dataclass
from google.genai import types
from gemini_client import client
from chunking import Chunk
from llm_utils import with_retry

VERIFIER_MODEL = "models/gemini-2.0-flash-lite"

@dataclass
class Claim:
    text:      str           # one sentence from the answer
    citations: list[int]     # the [n] numbers attached to it

@dataclass
class ClaimVerdict:
    claim:     Claim
    supported: bool
    reason:    str


def parse_claims(answer: str) -> list[Claim]:
    sentences = re.split(r'(?<=[.!?])\s+', answer.strip())
    claims = []
    for sentence in sentences:
        citations = [int(n) for n in re.findall(r'\[(\d+)\]', sentence)]
        if sentence.strip():
            claims.append(Claim(text=sentence.strip(), citations=citations))
    return claims

def verify_claim(claim_text: str, chunk: Chunk) -> bool:
    prompt = (
        "You are a strict fact-checker. Determine whether the CLAIM is fully "
        "supported by the PASSAGE. Answer with ONLY 'YES' or 'NO'.\n"
        "- Answer YES only if the passage directly states or clearly implies the claim.\n"
        "- Answer NO if the claim adds information not in the passage, even if it sounds plausible.\n\n"
        f"PASSAGE: {chunk.text}\n\n"
        f"CLAIM: {claim_text}\n\n"
        "Supported?"
    )
    response = with_retry(lambda: client.models.generate_content(
        model=VERIFIER_MODEL, contents=prompt,
        config=types.GenerateContentConfig(temperature=0, max_output_tokens=5),
    ))
    try:
        verdict = response.text.strip().upper()
        return verdict.startswith("YES")
    except (ValueError, AttributeError):
        return False

def _batch_verify(items: list[tuple[int, Claim, str]]) -> dict[int, bool]:
    """Verify many (claim, evidence) pairs in ONE LLM call.
    Returns {index: supported}. Missing/garbled lines default to False (fail-closed)."""
    blocks = [
        f"CLAIM {i}:\n{claim.text}\nEVIDENCE {i}:\n{evidence}"
        for i, claim, evidence in items
    ]
    prompt = (
        "You are a strict fact-checker. For each numbered CLAIM, decide whether it is FULLY "
        "supported by its EVIDENCE. Supported only if the evidence directly states or clearly "
        "implies the claim; if the claim adds information not in the evidence, answer NO.\n"
        "Respond with exactly one line per claim in the form '<number>: YES' or '<number>: NO'. "
        "No other text.\n\n"
        + "\n\n".join(blocks)
    )
    response = with_retry(lambda: client.models.generate_content(
        model=VERIFIER_MODEL, contents=prompt,
        config=types.GenerateContentConfig(temperature=0, max_output_tokens=200)))
    results: dict[int, bool] = {}
    try:
        for line in response.text.strip().splitlines():
            m = re.match(r"\s*(\d+)\s*:\s*(YES|NO)", line.strip(), re.IGNORECASE)
            if m:
                results[int(m.group(1))] = m.group(2).upper() == "YES"
    except (AttributeError, ValueError):
        pass
    return results


def verify_answer(answer: str, chunks: list[Chunk]) -> list[ClaimVerdict]:
    """Verify every claim in the answer against its cited chunk(s) in ONE batched LLM call."""
    claims   = parse_claims(answer)
    verdicts: list[ClaimVerdict | None] = [None] * len(claims)
    to_check: list[tuple[int, Claim, str]] = []

    for i, claim in enumerate(claims):
        if not claim.citations:
            verdicts[i] = ClaimVerdict(claim, supported=False, reason="no citation")
            continue
        evidence = "\n---\n".join(
            chunks[n - 1].text for n in claim.citations if 0 <= n - 1 < len(chunks)
        )
        if not evidence:
            verdicts[i] = ClaimVerdict(claim, supported=False, reason="cited passage not found")
            continue
        to_check.append((i, claim, evidence))

    if to_check:
        results = _batch_verify(to_check)
        for i, claim, _ in to_check:
            supported   = results.get(i, False)
            verdicts[i] = ClaimVerdict(
                claim, supported=supported,
                reason="supported" if supported else "not supported by cited passage",
            )
    return verdicts

import time
import numpy as np
from chunking import build_chunks, chunk_document, Chunk
from dense_index import build_index, embed_text, search as dense_search
from sparse_index import build_bm25_index, search as bm25_search
from fusion import reciprocal_rank_fusion
from rerank import rerank
from generation import generate_answer
from verification import verify_answer
from llm_utils import with_retry


class RagEngine:
    """Owns the live, mutable index state (chunks + dense matrix + bm25) and
    orchestrates the full pipeline. One instance lives for the server's lifetime."""

    def __init__(self):
        self.chunks              = build_chunks()
        self.matrix, self.chunks = build_index(self.chunks)
        self.bm25                = build_bm25_index(self.chunks)

    def query(self, query_text: str, top_k: int = 3) -> dict:
        timings = {}

        t0 = time.perf_counter()
        dense_results = dense_search(query_text, self.matrix, self.chunks, top_k=len(self.chunks))
        bm25_results  = bm25_search(query_text, self.bm25, self.chunks, top_k=len(self.chunks))
        fused         = reciprocal_rank_fusion([dense_results, bm25_results],
                                               top_k=max(top_k * 2, 5))
        timings["retrieval"] = round((time.perf_counter() - t0) * 1000)

        t0 = time.perf_counter()
        reranked           = rerank(query_text, fused, top_k=top_k)
        timings["rerank"]   = round((time.perf_counter() - t0) * 1000)

        used_chunks = [chunk for _, chunk in reranked]
        t0 = time.perf_counter()
        answer               = generate_answer(query_text, used_chunks)
        timings["generation"] = round((time.perf_counter() - t0) * 1000)

        t0 = time.perf_counter()
        verdicts                = verify_answer(answer, used_chunks)
        timings["verification"] = round((time.perf_counter() - t0) * 1000)

        claims = [
            {"text": v.claim.text, "citations": v.claim.citations,
             "supported": v.supported, "reason": v.reason}
            for v in verdicts
        ]
        sources = [
            {"n": i + 1, "source": chunk.source,
             "section": chunk.metadata.get("h2", ""),
             "text": chunk.text, "rerank_score": score}
            for i, (score, chunk) in enumerate(reranked)
        ]
        return {
            "query":      query_text,
            "answer":     answer,
            "claims":     claims,
            "sources":    sources,
            "timings_ms": timings,
        }

    def add_documents(self, filename: str, text: str) -> dict:
        """Chunk + embed a new document and merge it into the live index."""
        new_chunks = chunk_document(filename, text)
        if not new_chunks:
            return {"filename": filename, "chunks_added": 0,
                    "total_chunks": len(self.chunks)}

        new_vectors = [
            with_retry(lambda c=c: embed_text(c.text, task_type="RETRIEVAL_DOCUMENT"))
            for c in new_chunks
        ]
        new_matrix = np.array(new_vectors, dtype=np.float32)

        self.chunks = self.chunks + new_chunks
        self.matrix = np.vstack([self.matrix, new_matrix])
        self.bm25   = build_bm25_index(self.chunks)

        return {"filename": filename, "chunks_added": len(new_chunks),
                "total_chunks": len(self.chunks)}

        raise HTTPException(status_code=503, detail="Index not ready")

        raise HTTPException(status_code=400, detail="Only .md or .txt files are supported")

from dataclasses import dataclass, field

@dataclass
class EvalExample:
    query:              str
    relevant_chunk_ids: set[str]   # e.g. {"aurora_router_manual.md#3"}

EVAL_SET = [
    EvalExample("how do I fix error E-4021?",                  {"aurora_router_manual.md#3"}),
    EvalExample("what does a solid amber light mean?",            {"aurora_router_manual.md#2"}),
    EvalExample("my router can't get an IP address from the modem", {"aurora_router_manual.md#3"}),
    EvalExample("what is a mesh network?",                        {"wifi_concepts.md#1"}),
    EvalExample("why is the wifi weak in my back bedroom?",        {"wifi_concepts.md#0"}),
]

def chunk_id(chunk) -> str:
    return f"{chunk.source}#{chunk.chunk_index}"

def recall_at_k(results: list[tuple[float, "Chunk"]],
                relevant_ids: set[str], k: int) -> float:
    retrieved_ids = {chunk_id(chunk) for _, chunk in results[:k]}
    return 1.0 if retrieved_ids & relevant_ids else 0.0

def reciprocal_rank(results: list[tuple[float, "Chunk"]],
                    relevant_ids: set[str]) -> float:
    for rank, (_, chunk) in enumerate(results, start=1):
        if chunk_id(chunk) in relevant_ids:
            return 1.0 / rank
    return 0.0

def evaluate_retrieval(retriever, k: int = 5) -> tuple[float, float]:
    recalls, rrs = [], []
    for ex in EVAL_SET:
        results = retriever(ex.query)
        recalls.append(recall_at_k(results, ex.relevant_chunk_ids, k))
        rrs.append(reciprocal_rank(results, ex.relevant_chunk_ids))
    return sum(recalls) / len(recalls), sum(rrs) / len(rrs)

    chunks         = build_chunks()
    matrix, chunks = build_index(chunks)

    def dense_only(q):    return dense_search(q, matrix, chunks, top_k=9)
    def bm25_only(q):     return bm25_search(q, bm25, chunks, top_k=9)
    def hybrid(q):
        return reciprocal_rank_fusion(

    def hybrid_rerank(q): return rerank(q, hybrid(q)[:5], top_k=5)

    configs = {
        "Dense only":    dense_only,
        "BM25 only":     bm25_only,
        "Hybrid (RRF)":  hybrid,
        "Hybrid+Rerank": hybrid_rerank,
    }

    print(f"{'Configuration':<16} {'Recall@3':>9} {'MRR':>7}")
    print("-" * 34)
    for name, retriever in configs.items():
        recall, mrr = evaluate_retrieval(retriever, k=3)

generate_eval.py Eval LLM-generated synthetic eval set — saves to data/eval_set.json 
import json
from pathlib import Path
from google.genai import types
from gemini_client import client
from chunking import build_chunks, Chunk
from evaluate import EvalExample, chunk_id
from llm_utils import with_retry
import time


GEN_MODEL = "models/gemini-2.0-flash"
EVAL_PATH = Path(__file__).resolve().parent.parent / "data" / "eval_set.json"

def generate_questions(chunk: Chunk, n: int = 2) -> list[str]:
    prompt = (
        f"Read the passage below and write {n} distinct questions that it directly answers.\n"
        "Vary the style: make some use exact keywords from the passage, and some paraphrase "
        "the meaning in different words.\n"
        "Output ONLY the questions, one per line, with no numbering or extra text.\n\n"
        f"Passage:\n{chunk.text}"
    )
    response = with_retry(lambda: client.models.generate_content(
        model=GEN_MODEL, contents=prompt,
        config=types.GenerateContentConfig(temperature=0.7)))
    lines     = [line.strip() for line in response.text.strip().split("\n")]
    questions = [line for line in lines if line and line.endswith("?")]
    return questions

def build_synthetic_eval_set(chunks: list[Chunk], n: int = 2) -> list[EvalExample]:
    examples = []
    for chunk in chunks:
        for question in generate_questions(chunk, n):
            examples.append(EvalExample(
                query=question, relevant_chunk_ids={chunk_id(chunk)}))
        time.sleep(4)   # proactive throttle: stay under 15 req/min
    return examples


def save_eval_set(examples: list[EvalExample], path: Path = EVAL_PATH) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    data = [
        {"query": ex.query,
         "relevant_chunk_ids": sorted(ex.relevant_chunk_ids)}
        for ex in examples
    ]
    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

def load_eval_set(path: Path = EVAL_PATH) -> list[EvalExample]:
    with open(path, encoding="utf-8") as f:
        data = json.load(f)
    return [
        EvalExample(query=d["query"],
                    relevant_chunk_ids=set(d["relevant_chunk_ids"]))
        for d in data
    ]
